01. Timeline Breakdown

Animation Name: Page Intro Reveal Library: GSAP 
let pageLoadTimeline = gsap.timeline({
    onStart: () => {
        gsap.set(".transition", { 
            visibility: "visible", 
            background: "transparent"
        });
    },
    onComplete: () => {
        gsap.set(".transition", { visibility: "hidden" });
    },
    defaults: { ease: "power2.inOut" }
});

pageLoadTimeline.to(".transition-block", {
    opacity: 0,
    scale: 0.5,
    duration: 0.4,
    stagger: { amount: 0.8, from: "random" },
}, 0.2);
